                    AJAX (Legacy): XMLHttpRequest / jQuery
                    for maintenance
                
      
                
      
                    
      
                        “AJAX” is the concept of fetching data without reloading the page. Historically done with
                        XMLHttpRequest or $.ajax.
                    
      

                    
      
                        Example (read-only): XMLHttpRequest
                        
      const xhr = new XMLHttpRequest();
xhr.open('GET', '/data/users.json');
xhr.onload = () => {
  if (xhr.status === 200) {
    const data = JSON.parse(xhr.responseText);
    console.log(data);
  }
};
xhr.onerror = () => console.error('Network error');
xhr.send();
      
                    
      

                    
      
                        Example (read-only): jQuery $.ajax
                        
      $.ajax({
  url: 'users.json',
  dataType: 'json',
  success: function (data) { console.log(data); },
  error: function () { console.error('Error'); }
});
